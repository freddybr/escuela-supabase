CREATE TABLE public.grados (
  id bigint GENERATED ALWAYS AS IDENTITY NOT NULL,
  grado_numero text,
  grado_descripcion text,
  grado_nombre text NOT NULL,
  CONSTRAINT grados_pkey PRIMARY KEY (id)
);
CREATE TABLE public.materias (
  id bigint GENERATED ALWAYS AS IDENTITY NOT NULL,
  materia_nombre text NOT NULL,
  materia_descripcion text,
  CONSTRAINT materias_pkey PRIMARY KEY (id)
);
CREATE TABLE public.programas (
  id bigint GENERATED ALWAYS AS IDENTITY NOT NULL,
  materia_id bigint NOT NULL,
  programa_objetivo text,
  programa_tema text NOT NULL,
  programa_texto text,
  programa_estatus USER-DEFINED DEFAULT 'Elaborando'::programa_estatus,
  CONSTRAINT programas_pkey PRIMARY KEY (id),
  CONSTRAINT programas_materia_id_fkey FOREIGN KEY (materia_id) REFERENCES public.materias(id)
);
CREATE TABLE public.asignaciones (
  id bigint GENERATED ALWAYS AS IDENTITY NOT NULL,
  programa_id bigint NOT NULL,
  grado_id bigint NOT NULL,
  asigna_estatus USER-DEFINED DEFAULT 'Pendiente'::asigna_estatus,
  anio_id bigint,
  CONSTRAINT asignaciones_pkey PRIMARY KEY (id),
  CONSTRAINT asignaciones_programa_id_fkey FOREIGN KEY (programa_id) REFERENCES public.programas(id),
  CONSTRAINT asignaciones_grado_id_fkey FOREIGN KEY (grado_id) REFERENCES public.grados(id),
  CONSTRAINT asignaciones_anio_id_fkey FOREIGN KEY (anio_id) REFERENCES public.anio(id)
);
CREATE TABLE public.clases (
  id bigint GENERATED ALWAYS AS IDENTITY NOT NULL,
  programa_id bigint NOT NULL,
  clase_num integer,
  clase_tema text NOT NULL,
  clase_objetivo text,
  clase_texto text,
  CONSTRAINT clases_pkey PRIMARY KEY (id),
  CONSTRAINT clases_id_programa_fkey FOREIGN KEY (programa_id) REFERENCES public.programas(id)
);
CREATE TABLE public.anio (
  id bigint GENERATED ALWAYS AS IDENTITY NOT NULL,
  anio_periodo text,
  anio_inicio date,
  anio_fin date,
  CONSTRAINT anio_pkey PRIMARY KEY (id)
);
CREATE TABLE public.alumnos (
  id bigint NOT NULL,
  alumno_nombre text NOT NULL,
  alumno_email text,
  alumno_birthday date,
  alumno_sexo USER-DEFINED,
  alumno_representante text,
  alumno_telf text,
  alumno_direccion text,
  alumno_imagen_url text,
  grado_id bigint,
  CONSTRAINT alumnos_pkey PRIMARY KEY (id),
  CONSTRAINT alumnos_grado_id_fkey FOREIGN KEY (grado_id) REFERENCES public.grados(id)
);
CREATE TABLE public.profesores (
  id bigint NOT NULL,
  profe_nombre text NOT NULL,
  profe_email text,
  profe_telf text,
  profe_estatus text,
  profe_rol text,
  profe_imagen_url text,
  grado_id bigint,
  CONSTRAINT profesores_pkey PRIMARY KEY (id),
  CONSTRAINT profesores_grado_id_fkey FOREIGN KEY (grado_id) REFERENCES public.grados(id)
);
CREATE TABLE public.control (
  id bigint GENERATED ALWAYS AS IDENTITY NOT NULL,
  asigna_id bigint NOT NULL,
  control_fecha date,
  profe_id bigint,
  control_observ text,
  clase_id bigint,
  control_estatus USER-DEFINED DEFAULT 'Pendiente'::control_estatus,
  CONSTRAINT control_pkey PRIMARY KEY (id),
  CONSTRAINT control_profe_id_fkey FOREIGN KEY (profe_id) REFERENCES public.profesores(id),
  CONSTRAINT control_clase_id_fkey FOREIGN KEY (clase_id) REFERENCES public.clases(id),
  CONSTRAINT control_asigna_id_fkey FOREIGN KEY (asigna_id) REFERENCES public.asignaciones(id)
);
CREATE TABLE public.asistencias (
  id bigint GENERATED ALWAYS AS IDENTITY NOT NULL,
  control_id bigint NOT NULL,
  alumno_id bigint NOT NULL,
  asist_presente boolean NOT NULL DEFAULT false,
  asist_evaluacion USER-DEFINED,
  asist_observacion text,
  CONSTRAINT asistencias_pkey PRIMARY KEY (id),
  CONSTRAINT fk_asistencias_alumno FOREIGN KEY (alumno_id) REFERENCES public.alumnos(id),
  CONSTRAINT fk_asistencias_control FOREIGN KEY (control_id) REFERENCES public.control(id)
);
